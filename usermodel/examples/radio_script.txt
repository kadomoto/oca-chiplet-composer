# Send four words on channel 3 at -35 dBm and read the link statistics
write CHANNEL 3
write TX_POWER_DBM -35
write TX_FIFO 0x11223344
write TX_FIFO 0x55667788
write TX_FIFO 0x99AABBCC
write TX_FIFO 0xDDEEFF00
write CTRL START
poll STATUS.DONE
read TX_COUNT
read AIRTIME_NS
read RSSI_DBM
read SNR_CDB
read BER_PPM
expect STATUS.BUSY 0
